Step 1 — Add call_routing_responses table to Supabase
===================================================================

Run this migration in Supabase SQL Editor:

CREATE TABLE IF NOT EXISTS call_routing_responses (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id      uuid REFERENCES companies(id) ON DELETE CASCADE,
  biggest_focus   text[],
  lead_sources    text[],
  software_usage  text,
  account_count   text,
  call_notes      text,
  booked_at       timestamptz DEFAULT now(),
  call_scheduled  text
);

CREATE INDEX IF NOT EXISTS routing_company_idx 
  ON call_routing_responses(company_id);

  ADD COLUMN IF NOT EXISTS call_booked boolean DEFAULT false;

'use client'

import { useEffect, Suspense } from 'react'
import { useSearchParams } from 'next/navigation'
import Header from '@/components/Header'

function BookContent() {
  const searchParams = useSearchParams()
  const companyId = searchParams.get('company_id') || ''
  const name = searchParams.get('name') || ''
  const email = searchParams.get('email') || ''

  useEffect(() => {
    // Load Calendly widget script
    const script = document.createElement('script')
    script.src = 'https://assets.calendly.com/assets/external/widget.js'
    script.async = true
    document.body.appendChild(script)
    return () => { document.body.removeChild(script) }
  }, [])

  // Build Calendly URL with prefill and passthrough param
  const calendlyUrl = new URL(process.env.NEXT_PUBLIC_CALENDLY_URL || 
    'https://calendly.com/your-username/discovery-call')
  calendlyUrl.searchParams.set('name', name)
  calendlyUrl.searchParams.set('email', email)
  // Pass company_id through so confirmation page knows who booked
  calendlyUrl.searchParams.set('a1', companyId)

  return (
    <div className="min-h-screen flex flex-col bg-gray-50">
      <Header />
      <main className="flex-1 py-10 px-4">
        <div className="max-w-2xl mx-auto">
          <div className="text-center mb-8">
            <div className="w-12 h-12 bg-green-100 rounded-full flex 
              items-center justify-center mx-auto mb-4">
              <svg className="w-6 h-6 text-green-600" fill="none" 
                stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" 
                  strokeWidth={2} d="M5 13l4 4L19 7" />
              </svg>
            </div>
            <h1 className="text-2xl font-bold text-navy mb-2">
              Your listing is live
            </h1>
            <p className="text-gray-500 text-sm max-w-md mx-auto">
              Grab a time below — we'll confirm your listing, turn on 
              referral distribution for your area, and show you leads 
              available right now. Takes 10 minutes.
            </p>
          </div>

          {/* Calendly inline embed */}
          <div 
            className="calendly-inline-widget bg-white rounded-xl 
              border border-gray-200 shadow-sm overflow-hidden"
            data-url={calendlyUrl.toString()}
            style={{ minWidth: '320px', height: '700px' }}
          />

          <p className="text-center text-xs text-gray-400 mt-4">
            Prefer to skip for now?{' '}
            <a href="/dashboard" className="text-accent hover:underline">
              Go to your dashboard →
            </a>
          </p>
        </div>
      </main>
    </div>
  )
}

import { useState, Suspense } from 'react'
import { useSearchParams, useRouter } from 'next/navigation'
import Header from '@/components/Header'

const FOCUS_OPTIONS = [
  'More commercial contracts',
  'Following up with leads faster',
  'Reducing missed calls',
  'Managing operations',
  'Cash flow challenges',
  'Preparing to scale or sell',
]

const SOURCE_OPTIONS = [
  'Referrals',
  'Google / SEO',
  'Paid ads',
  'Not consistently — that\'s the problem',
]

const SOFTWARE_OPTIONS = [
  'Yes, works well',
  'Yes but it\'s messy',
  'No — mostly manual',
]

const ACCOUNT_OPTIONS = [
  '1–5',
  '6–20',
  '21–50',
  '50+',
]

function ConfirmationContent() {
  const searchParams = useSearchParams()
  const router = useRouter()
  const companyId = searchParams.get('company_id') || ''

  const [form, setForm] = useState({
    biggest_focus: [] as string[],
    lead_sources: [] as string[],
    software_usage: '',
    account_count: '',
    call_notes: '',
  })

  const [done, setDone] = useState(false)

    arr.includes(val) ? arr.filter(x => x !== val) : [...arr, val]

    setSubmitting(true)
    await fetch('/api/claim/routing', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ...form, company_id: companyId }),

    setSubmitting(false)

    label, selected, onClick, multi

    onClick: () => void; multi?: boolean

      type="button"
      onClick={onClick}
      className={`p-3 rounded-lg border-2 text-left text-sm

        selected
          ? 'border-navy bg-navy/5 text-navy'
          : 'border-gray-200 hover:border-gray-300 text-gray-700'
      }`}
    >

        <span className={`inline-block w-4 h-4 rounded mr-2 
          border-2 align-middle ${
          selected 
            ? 'bg-navy border-navy' 
            : 'border-gray-400'
        }`} />
      )}
      {label}
    </button>
  )

  if (done) {
    return (
      <div className="min-h-screen flex flex-col bg-gray-50">
        <Header />
        <main className="flex-1 flex items-center justify-center px-4">
          <div className="max-w-md text-center">
            <div className="w-16 h-16 bg-green-100 rounded-full flex 
              items-center justify-center mx-auto mb-4">
              <svg className="w-8 h-8 text-green-600" fill="none" 
                stroke="currentColor" viewBox="0 0 24 24">

                  strokeWidth={2} d="M5 13l4 4L19 7" />
              </svg>
            </div>
            <h1 className="text-2xl font-bold text-navy mb-2">
              You're all set
            </h1>
            <p className="text-gray-500 mb-6">
              You'll get a reminder before the call. 
              In the meantime, your dashboard shows leads 
              available in your area right now.
            </p>
            <a href="/dashboard" 
              className="inline-block bg-navy text-white font-semibold

  return (

      <Header />

            >
              {submitting ? 'Saving…' : 'Submit →'}

          </div>
        </div>
      </main>
    </div>
  )
}

export default function ConfirmationPage() {

Step 6 — Build the /api/claim/routing API route
===================================================================

Create app/api/claim/routing/route.ts:

import { NextRequest, NextResponse } from 'next/server'
import { supabaseAdmin } from '@/lib/supabase'
import { sendEmail } from '@/lib/resend'

export async function POST(req: NextRequest) {
  const body = await req.json()
  const { 
    company_id, biggest_focus, lead_sources, 
    software_usage, account_count, call_notes 
  } = body

  if (!company_id) {
    return NextResponse.json({ error: 'Missing company_id' }, 
      { status: 400 })
  }

  await supabaseAdmin.from('call_routing_responses').insert({
    company_id,
    biggest_focus,
    lead_sources,
    software_usage,
    account_count,
    call_notes,
  })

  // Mark as call_booked in onboarding
  await supabaseAdmin
    .from('company_onboarding')
    .update({ call_booked: true })
    .eq('company_id', company_id)

  // Send admin notification with routing data
  const { data: company } = await supabaseAdmin
    .from('companies')
    .select('name, city, state, email')
    .eq('id', company_id)
    .single()

  if (company) {
    const agenda = biggest_focus?.includes('Cash flow challenges')
      ? 'LENDING REFERRAL'
      : biggest_focus?.includes('Preparing to scale or sell')
      ? 'ACQUISITION CONVERSATION'
      : 'GHL + LEADS PITCH'

    await sendEmail(
      process.env.ADMIN_EMAIL!,
      `📞 Call booked — ${company.name} | Agenda: ${agenda}`,
      `<div style="font-family:sans-serif;max-width:600px">
        <h2>Call Booked — ${company.name}</h2>
        <p><strong>City:</strong> ${company.city}, ${company.state}</p>
        <p><strong>Email:</strong> ${company.email}</p>
        <hr/>
        <h3>Your Agenda: ${agenda}</h3>
        <p><strong>Biggest Focus:</strong> ${biggest_focus?.join(', ')}</p>
        <p><strong>Current Lead Sources:</strong> ${lead_sources?.join(', ')}</p>
        <p><strong>Software:</strong> ${software_usage}</p>
        <p><strong>Active Accounts:</strong> ${account_count}</p>
        ${call_notes ? `<p><strong>Their Notes:</strong> ${call_notes}</p>` : ''}
        <hr/>
        <p style="color:#666;font-size:13px">
          Open with: "Before we get into anything — you mentioned 
          [${biggest_focus?.[0]}]. Tell me more about that."
        </p>
      </div>`
    )
  }

  return NextResponse.json({ success: true })
}
