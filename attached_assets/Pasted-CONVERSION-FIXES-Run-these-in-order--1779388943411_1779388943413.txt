Fix C6 — Fix claim search to query by name, not slug
===================================================================

In app/api/company-by-slug/route.ts, the current GET handler 
finds a company by slug. It needs to also handle a name/city search
for the claim search page.

Check the current route — if it only queries by exact slug match,
update it to handle a 'q' query param as a name search:

export async function GET(req: NextRequest) {
  const slug = req.nextUrl.searchParams.get('slug')
  const q = req.nextUrl.searchParams.get('q')
  
  if (slug) {
    // Existing slug lookup for company profile page
    const { data } = await supabaseAdmin
      .from('companies')
      .select('id, name, city, state, slug, claimed')
      .eq('slug', slug)
      .single()
    return NextResponse.json({ company: data })
  }
  
  if (q) {
    // Name search for claim flow
    const { data } = await supabaseAdmin
      .from('companies')
      .select('id, name, city, state, slug, claimed')
      .ilike('name', `%${q}%`)
      .eq('active', true)
      .eq('claimed', false)  // Only show unclaimed companies
      .order('rating', { ascending: false })
      .limit(10)
    return NextResponse.json(data || [])
  }
  
  return NextResponse.json({ error: 'Missing slug or q param' }, 
    { status: 400 })
}
