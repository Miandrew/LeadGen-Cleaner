export async function generateStaticParams() {
  // All 50 state pages
  const stateParams = US_STATES.map((s) => ({
    location: s.name.toLowerCase().replace(/\s+/g, '-'),
  }))

  // Top city pages from actual database
  try {
    const { data } = await supabaseAdmin
      .from('companies')
      .select('city, state')
      .eq('active', true)
      .not('city', 'is', null)
      .not('state', 'is', null)
      .limit(2000)

    const seen = new Set<string>()
    const cityParams: { location: string }[] = []

    for (const c of data || []) {
      if (!c.city || !c.state) continue
      const slug = `${c.city.toLowerCase().replace(/\s+/g, '-')}-${c.state.toLowerCase()}`
      if (!seen.has(slug)) {
        seen.add(slug)
        cityParams.push({ location: slug })
      }
      if (cityParams.length >= 500) break
    }

    return [...stateParams, ...cityParams]
  } catch {
    return stateParams
  }
}

export async function generateStaticParams() {
  // All service + state combinations
  const params: { service: string; location: string }[] = []
  
  for (const service of SERVICE_TYPES) {
    // State pages for each service
    for (const state of US_STATES) {
      params.push({
        service: service.value,
        location: state.name.toLowerCase().replace(/\s+/g, '-'),
      })
    }
  }
  
  // Top service + city combinations from database
  try {
    const { data } = await supabaseAdmin
      .from('companies')
      .select('city, state, services')
      .eq('active', true)
      .not('city', 'is', null)
      .limit(3000)
    
    const seen = new Set<string>()
    let count = 0
    
    for (const company of data || []) {
      if (!company.city || !company.state) continue
      for (const svc of company.services || []) {
        const match = SERVICE_TYPES.find(s => s.value === svc)
        if (!match) continue
        const citySlug = `${company.city.toLowerCase().replace(/\s+/g, '-')}-${company.state.toLowerCase()}`
        const key = `${svc}:${citySlug}`
        if (!seen.has(key)) {
          seen.add(key)
          params.push({ service: svc, location: citySlug })
          count++
        }
        if (count >= 1000) break
      }
      if (count >= 1000) break
    }
  } catch {}
  
  return params
}

function parseCityState(location: string): { city: string; state: string } {
  const parts = location.split('-')
  const state = parts[parts.length - 1].toUpperCase()
  const city = parts.slice(0, -1).join(' ')
    .replace(/\b\w/g, (c) => c.toUpperCase())
  return { city, state }
}

export async function generateMetadata({ params }: { params: { location: string } }): Promise<Metadata> {

}

Fix P6 — Fix the sitemap to include all pSEO URL types
===================================================================

Replace the entire content of app/sitemap.xml/route.ts with:

import { NextResponse } from 'next/server'
import { supabaseAdmin } from '@/lib/supabase'
import { US_STATES, SERVICE_TYPES } from '@/lib/utils'

export const revalidate = 86400

export async function GET() {
  const site = 'https://commercialcleaningnearme.com'

  const [{ data: companies }, { data: locationData }] = 
    await Promise.all([
      supabaseAdmin
        .from('companies')
        .select('slug')
        .eq('active', true)
        .limit(10000),
      supabaseAdmin
        .from('companies')
        .select('city, state')
        .eq('active', true)
        .limit(2000),
    ])

  // Build unique city slugs
  const citySet = new Set<string>()
  for (const c of locationData || []) {
    if (c.city && c.state) {
      citySet.add(
        `${c.city.toLowerCase().replace(/\s+/g, '-')}-${c.state.toLowerCase()}`
      )
    }
  }
  const citySlugs = Array.from(citySet).slice(0, 1000)

  const urls: string[] = []

  const url = (loc: string, freq: string, pri: string) =>
    `  <url><loc>${site}${loc}</loc><changefreq>${freq}</changefreq><priority>${pri}</priority></url>`

  // Static pages
  urls.push(url('/', 'daily', '1.0'))
  urls.push(url('/search', 'weekly', '0.9'))
  urls.push(url('/claim', 'weekly', '0.8'))
  urls.push(url('/browse-by-service', 'weekly', '0.8'))
  urls.push(url('/browse-by-state', 'weekly', '0.8'))

  // Company profiles
  for (const c of companies || []) {
    if (c.slug) urls.push(url(`/company/${c.slug}`, 'monthly', '0.6'))
  }

  // commercial-cleaning state pages
  for (const s of US_STATES) {
    const slug = s.name.toLowerCase().replace(/\s+/g, '-')
    urls.push(url(`/commercial-cleaning/${slug}`, 'weekly', '0.8'))
    urls.push(url(`/state/${slug}`, 'weekly', '0.7'))
  }

  // commercial-cleaning city pages
  for (const slug of citySlugs) {
    urls.push(url(`/commercial-cleaning/${slug}`, 'weekly', '0.7'))
    urls.push(url(`/commercial-cleaning-near-me/${slug}`, 'weekly', '0.7'))
  }

  // service hub pages
  for (const svc of SERVICE_TYPES) {
    urls.push(url(`/service/${svc.value}`, 'weekly', '0.8'))
    // service + state pages
    for (const s of US_STATES) {
      const stateSlug = s.name.toLowerCase().replace(/\s+/g, '-')
      urls.push(url(`/service/${svc.value}/${stateSlug}`, 'weekly', '0.7'))
    }
    // service + city pages
    for (const slug of citySlugs.slice(0, 300)) {
      urls.push(url(`/service/${svc.value}/${slug}`, 'weekly', '0.6'))
    }
  }

  const xml = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${urls.join('\n')}
</urlset>`

  return new NextResponse(xml, {
    headers: { 'Content-Type': 'application/xml' },
  })
}

hub that shows "Office Cleaning in Texas (450 companies)" etc. It

To:

company listing page, /state/texas is the service directory page.
