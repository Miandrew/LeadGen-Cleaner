Create app/api/auth/logout/route.ts:

import { NextResponse } from 'next/server'
import { createSupabaseServerClient, isSupabaseConfigured } from '@/lib/supabase'
import { cookies } from 'next/headers'

export async function POST() {
  if (isSupabaseConfigured()) {
    const cookieStore = cookies()
    const supabase = createSupabaseServerClient(
      (name) => cookieStore.get(name)?.value
    )
    if (supabase) await supabase.auth.signOut()
  }
  const res = NextResponse.redirect(
    new URL('/login', process.env.NEXT_PUBLIC_SITE_URL)
  )
  return res
}

Fix H — Fix pSEO city pages to actually pre-render (critical for SEO)
===================================================================

The current commercial-cleaning/[location] route uses

dynamically rendered page — not a pre-built static page. This

In app/commercial-cleaning/[location]/page.tsx:
1. Remove: export const dynamic = 'force-dynamic'
2. Replace generateStaticParams with one that generates both state 
   AND top city pages:

export async function generateStaticParams() {
  // Always include all 50 state pages
  const stateParams = US_STATES.map((s) => ({
    location: s.name.toLowerCase().replace(/\s+/g, '-'),
  }))

  // Add top 200 city combinations from actual company data
  try {
    const { data } = await supabaseAdmin
      .from('companies')
      .select('city, state')
      .eq('active', true)
      .not('city', 'is', null)
      .not('state', 'is', null)
      .limit(1000)

    const citySet = new Set<string>()
    data?.forEach((c) => {
      if (c.city && c.state) {
        const slug = `${c.city.toLowerCase().replace(/\s+/g, '-')}-${c.state.toLowerCase()}`
        citySet.add(slug)
      }
    })

    const cityParams = Array.from(citySet)
      .slice(0, 200)
      .map((slug) => ({ location: slug }))

    return [...stateParams, ...cityParams]
  } catch {
    return stateParams
  }
}
